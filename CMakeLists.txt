cmake_minimum_required(VERSION 3.16)
project(algorithm_practice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(practice INTERFACE)
target_include_directories(practice INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(algorithm_tests tests/test_algorithms.cpp)
target_link_libraries(algorithm_tests PRIVATE practice)
add_executable(route_example examples/route_cost.cpp)
target_link_libraries(route_example PRIVATE practice)
enable_testing()
add_test(NAME algorithm_tests COMMAND algorithm_tests)
